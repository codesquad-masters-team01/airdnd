// 1. 회원 테이블 (게스트, 호스트, 관리자 공통)
Table members {
  id bigint [pk, increment, note: 'Surrogate Key']
  email varchar(255) [not null, unique]
  nickname varchar(100) [not null]
  role varchar(20) [not null, note: 'GUEST, HOST, ADMIN']
  oauth_provider varchar(20) [not null, note: 'GITHUB, GOOGLE']
  oauth_id varchar(255) [not null, unique]
  is_deleted boolean [not null, default: false, note: '삭제(탈퇴) 여부']
}

// 2. 숙소 테이블
Table rooms {
  id bigint [pk, increment]
  host_id bigint [not null, ref: > members.id]
  name varchar(255) [not null]
  description text

  // 🌍 주소는 사람이 읽는 텍스트로만 보관
  address varchar(500) [not null, note: '전체 주소 텍스트']
  country_code varchar(5) [not null, note: 'ISO 국가 코드 (예: KR, US, FR)']

  // 📍 지도 매핑 및 백엔드 반경 검색의 유일한 기준 (충돌 위험 없음)
  latitude decimal(15, 12) [not null, note: '위도']
  longitude decimal(15, 12) [not null, note: '경도']

  price_per_night int [not null]
  max_capacity int [not null]
  allows_infants boolean [not null, default: true]
  allows_pets boolean [not null, default: false, note: '반려동물 동반 가능 여부']

  is_active boolean [not null, default: true, note: 'true: 노출, false: 숨김']
  is_deleted boolean [not null, default: false, note: 'Soft Delete']
}

// 3. 숙소 이미지 테이블 (1:N)
Table room_images {
  id bigint [pk, increment]
  room_id bigint [not null, ref: > rooms.id]
  image_url varchar(500) [not null]
  is_representative boolean [not null, default: false, note: '대표 이미지 여부']
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
}

// 4. 예약 테이블
Table reservations {
  id bigint [pk, increment]
  guest_id bigint [not null, ref: > members.id]
  room_id bigint [not null, ref: > rooms.id]
  check_in_date date [not null]
  check_out_date date [not null]
  total_price int [not null]

  // 👥 예약 인원 및 옵션 세분화
  adult_count int [not null, default: 1, note: '성인 수']
  child_count int [not null, default: 0, note: '어린이 수']
  infant_count int [not null, default: 0, note: '유아 수']
  has_pets boolean [not null, default: false, note: '반려동물 동반 여부']

  status varchar(20) [not null, note: 'PENDING, CONFIRMED, CANCELED']
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`]
  deleted_at timestamp [note: 'Soft Delete']
}

// 5. 리뷰 테이블 (예약과 1:1 관계)
Table reviews {
  id bigint [pk, increment]
  reservation_id bigint [not null, ref: - reservations.id]
  rating int [not null, note: '1 ~ 5점']
  comment text
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`]
  deleted_at timestamp [note: 'Soft Delete']
}

// 6-1. 위시리스트 폴더 테이블 (회원가입 시 "가고 싶은 곳" 기본 자동 생성전략 반영)
Table wishlists {
  id bigint [pk, increment, note: 'Surrogate Key']
  member_id bigint [not null, ref: > members.id]
  name varchar(100) [not null, note: '폴더 이름 (기본값: 가고 싶은 곳)']
  created_at timestamp [default: `CURRENT_TIMESTAMP`, note: '최신순 정렬용']
}

// 6-2. 위시리스트-숙소 매핑 테이블 (N:M 중간 테이블 대체, 대리 키 전략)
Table wishlist_rooms {
  id bigint [pk, increment, note: 'Surrogate Key']
  wishlist_id bigint [not null, ref: > wishlists.id]
  room_id bigint [not null, ref: > rooms.id]
  created_at timestamp [default: `CURRENT_TIMESTAMP`, note: '최신 찜한순 정렬용']
}
